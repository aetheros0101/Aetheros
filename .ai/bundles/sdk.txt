use serde::{Deserialize, Serialize};
use uuid::Uuid;

#[derive(Debug, thiserror::Error)]
pub enum ClientError {
    #[error("HTTP error: {0}")]
    Http(#[from] reqwest::Error),
    #[error("API error: {status} — {message}")]
    Api { status: u16, message: String },
    #[error("Parse error: {0}")]
    Parse(String),
}

// ── Request modelleri ─────────────────────────────────────

#[derive(Debug, Serialize)]
pub struct SubmitTaskRequest {
    pub entrypoint: String,
    pub wasm_module_hex: String,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub priority: Option<String>,
    pub timeout_ms: u64,
    pub max_attempts: u32,
}

#[derive(Debug, Serialize)]
pub struct StartAgentRequest {
    pub objective: String,
    pub max_steps: usize,
    pub max_tokens: usize,
}

#[derive(Debug, Serialize)]
pub struct SubmitWorkflowRequest {
    pub name: String,
}

// ── Response modelleri ────────────────────────────────────

#[derive(Debug, Deserialize)]
pub struct SubmitTaskResponse {
    pub task_id: Uuid,
    pub status: String,
}

#[derive(Debug, Deserialize)]
pub struct TaskStateResponse {
    pub task_id: Uuid,
    pub state: String,
    pub attempts: u32,
}

#[derive(Debug, Deserialize)]
pub struct AgentResponse {
    pub execution_id: Uuid,
    pub status: String,
}

#[derive(Debug, Deserialize)]
pub struct WorkflowResponse {
    pub workflow_id: Uuid,
    pub status: String,
}

#[derive(Debug, Deserialize)]
pub struct HealthResponse {
    pub healthy: bool,
    pub version: String,
}

// ── Client ────────────────────────────────────────────────

pub struct AetherClient {
    endpoint: String,
    client: reqwest::Client,
}

impl AetherClient {
    pub fn new(endpoint: impl Into<String>) -> Self {
        Self {
            endpoint: endpoint
                .into()
                .trim_end_matches('/')
                .to_string(),
            client: reqwest::Client::new(),
        }
    }

    /// POST /tasks → task submit
    pub async fn submit_task(
        &self,
        req: SubmitTaskRequest,
    ) -> Result<SubmitTaskResponse, ClientError> {
        let resp = self
            .client
            .post(format!("{}/tasks", self.endpoint))
            .json(&req)
            .send()
            .await?;

        self.parse(resp).await
    }

    /// GET /tasks/:id → task state
    pub async fn task_state(
        &self,
        task_id: Uuid,
    ) -> Result<TaskStateResponse, ClientError> {
        let resp = self
            .client
            .get(format!(
                "{}/tasks/{}",
                self.endpoint, task_id
            ))
            .send()
            .await?;

        self.parse(resp).await
    }

    /// POST /tasks/:id/cancel
    pub async fn cancel_task(
        &self,
        task_id: Uuid,
    ) -> Result<serde_json::Value, ClientError> {
        let resp = self
            .client
            .post(format!(
                "{}/tasks/{}/cancel",
                self.endpoint, task_id
            ))
            .send()
            .await?;

        self.parse(resp).await
    }

    /// POST /agents → agent başlat
    pub async fn start_agent(
        &self,
        objective: impl Into<String>,
        max_steps: usize,
        max_tokens: usize,
    ) -> Result<AgentResponse, ClientError> {
        let req = StartAgentRequest {
            objective: objective.into(),
            max_steps,
            max_tokens,
        };

        let resp = self
            .client
            .post(format!("{}/agents", self.endpoint))
            .json(&req)
            .send()
            .await?;

        self.parse(resp).await
    }

    /// POST /workflows → workflow submit
    pub async fn submit_workflow(
        &self,
        name: impl Into<String>,
    ) -> Result<WorkflowResponse, ClientError> {
        let req = SubmitWorkflowRequest {
            name: name.into(),
        };

        let resp = self
            .client
            .post(format!(
                "{}/workflows",
                self.endpoint
            ))
            .json(&req)
            .send()
            .await?;

        self.parse(resp).await
    }

    /// GET /health
    pub async fn health(
        &self,
    ) -> Result<HealthResponse, ClientError> {
        let resp = self
            .client
            .get(format!("{}/health", self.endpoint))
            .send()
            .await?;

        self.parse(resp).await
    }

    async fn parse<T: for<'de> Deserialize<'de>>(
        &self,
        resp: reqwest::Response,
    ) -> Result<T, ClientError> {
        let status = resp.status();

        if !status.is_success() {
            let msg = resp
                .text()
                .await
                .unwrap_or_default();
            return Err(ClientError::Api {
                status: status.as_u16(),
                message: msg,
            });
        }

        resp.json::<T>().await.map_err(ClientError::Http)
    }
}
